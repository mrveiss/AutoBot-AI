# Single source of truth for shared Python dependency versions (#10524).
#
# Every component requirements file references this via:
#     -c <relative-path>/constraints/shared.txt
# Change a version HERE, once. NEVER re-declare a constrained library's version in a
# component requirements file — a CI guard (constraint-drift workflow) fails any PR
# that does, and pip would otherwise resolve different versions per component.
#
# Why a single version is mandatory (not just hygiene): Ansible can co-locate or
# relocate roles (backend, tts-worker, npu-worker, ai-stack) onto one host. Two roles
# with conflicting pins in the same environment make pip unsatisfiable, and numpy ABI
# skew corrupts .npy / pickled embeddings written by one role and read by another.

# numpy unified at 2.x across all roles after graspologic (numpy<2) was removed (#10524).
numpy>=2.5.4,<3.0.0

# openvino floor -- was restated as a literal in 7 independent files (requirements
# files, ansible roles/playbook, a group_var, a setup.py extra and install.sh), so
# every dependabot bump failed CI until the other six were hand-edited (#15408).
# `openvino-dev` has no release compatible with this floor -- see #14447/#14452/#14453.
openvino>=2026.4.1

# croniter: the cron engine behind the shared 5-field validator, the backend triggers, the LLC
# routine scheduler and the SLM schedule executor (#18449). Was declared in two files.
croniter>=6.2.4

# chromadb: the vector store client in the backend, the ai-stack container and code_analysis,
# plus the standalone server venvs the ansible redis role and install-bare-metal.sh build.
# Was declared in 6 places with 4 specs, one a live conflict (>=0.5) (#19478). The cap
# guards against breaking API changes; 1.5.5+ regenerated its protos for protobuf 5.x (#4053).
chromadb>=1.5.9,<2.0.0

# filelock: runtime dependency via torch (unbounded `filelock` requirement) and the HF stack.
# Floor at the release fixing CVE-2026-112940, CVE-2026-112913 and CVE-2026-112907 (#19426).
# Was also a stale 3.20.1 floor in ansible dependency_patching; that restatement is removed.
filelock>=4.2.4

# starlette: the security floor for the O(n^2) Range-header DoS. fastapi itself only requires
# starlette>=0.46.0, so its floor does not imply this one and it is declared here for every manifest.
starlette>=1.9.0

# fastapi: the shared floor.
# Was restated in 11 places at two versions (>=0.147.0 in 8 manifests, ==0.147.0 in
# requirements-ci/framework.txt, 0.147.0 in dependency_patching and an inline ansible spec), so every
# dependabot bump left the others behind (#18060). 0.162.0 changes how an omitted bool Form field
# is parsed; every bool Form parameter therefore carries an explicit default.
fastapi>=0.162.0

# torch / torchvision: ONE coupled pair (#19330, maintainer decision 2026-10-26). torchvision 0.29.x
# is built against torch 2.14.x -- move them together, never one alone. Was pinned in 7 places at
# three patch levels (2.12.1 in the backend image, #19072; 2.14.0 on GPU hosts; 2.14.1 elsewhere).
# `==2.14.1` also matches the CPU index's `2.14.1+cpu` and CUDA builds' `2.14.1+cuNNN` local versions.
torch==2.14.1
torchvision==0.29.1
